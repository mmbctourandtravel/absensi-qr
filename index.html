function cariAbsenTerbuka(sheet, idStaff) { const data = sheet.getDataRange().getValues(); const now = new Date(); const tglHariIni = Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd"); for (let i = data.length - 1; i >= 1; i--) { let rowId = String(data[i][1]).trim(); let jamMasuk = data[i][3]; // Kolom D (Jam Masuk) let jamPulang = data[i][4]; // Kolom E (Jam Pulang) let tglAbsenVal = data[i][0]; // Normalisasi tanggal spreadsheet ke string "yyyy-MM-dd" let tglAbsen = ""; if (tglAbsenVal instanceof Date) { tglAbsen = Utilities.formatDate(tglAbsenVal, TIMEZONE, "yyyy-MM-dd"); } else { tglAbsen = String(tglAbsenVal).trim(); if (tglAbsen.length > 10) tglAbsen = tglAbsen.substring(0, 10); } // SYARAT KETAT BARU: // Absen terbuka untuk pulang HANYA jika ID sama, jam pulang kosong, jam masuk ada, DAN TANGGALNYA HARUS HARI INI JUGA. // Jika sudah beda hari (kemarin lupa absen), tidak boleh dianggap absen pulang lagi. if (rowId === idStaff && String(jamPulang).trim() === "" && jamMasuk !== "") { if (tglAbsen === tglHariIni) { return i + 1; // Ketemu baris absen masuk hari ini yang belum pulang } } } return -1; // Tidak ada absen terbuka hari ini, artinya aman untuk buat Absen Datang baru! }
